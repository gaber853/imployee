The classes behind it 
class Employee {                       // abstract base
  constructor(name, id) { this.name = name; this.id = id; }
  calcPay() { throw new Error("override me"); }
}

class FixedSalary extends Employee {
  constructor(name, id, salary, bonus) { super(name, id); this.salary = salary; this.bonus = bonus; }
  calcPay() { return this.salary + this.bonus; }
}

class HourlySalary extends Employee {
  constructor(name, id, hours, rate) { super(name, id); this.hours = hours; this.rate = rate; }
  calcPay() { return this.hours * this.rate; }
}

// one loop, two behaviours (polymorphism)
const total = staff.reduce((sum, e) => sum + e.calcPay(), 0);
